# =====================
# Imports
# =====================
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_absolute_error, mean_squared_error
from statsmodels.tsa.api import VAR
import warnings
warnings.filterwarnings("ignore")

# =====================
# 1. DATA GENERATION
# =====================
def generate_data(n=1200, seed=42):
    np.random.seed(seed)
    t = np.arange(n)
    return pd.DataFrame({
        "energy": np.sin(0.02 * t) + np.random.normal(0, 0.05, n),
        "temperature": np.cos(0.015 * t),
        "humidity": np.sin(0.01 * t)
    })

data = generate_data()

# =====================
# 2. SCALING
# =====================
scaler = MinMaxScaler()
scaled = scaler.fit_transform(data)

# =====================
# 3. SEQUENCE BUILDER
# =====================
def build_sequences(arr, input_len=30, horizon=10):
    X, y = [], []
    for i in range(len(arr) - input_len - horizon):
        X.append(arr[i:i+input_len])
        y.append(arr[i+input_len:i+input_len+horizon])
    return np.array(X), np.array(y)

INPUT_LEN, HORIZON = 30, 10
X, y = build_sequences(scaled, INPUT_LEN, HORIZON)

split = int(0.8 * len(X))
X_train, X_test = X[:split], X[split:]
y_train, y_test = y[:split], y[split:]

decoder_train = np.concatenate([np.zeros_like(y_train[:, :1]), y_train[:, :-1]], axis=1)
decoder_test  = np.concatenate([np.zeros_like(y_test[:, :1]),  y_test[:, :-1]], axis=1)

# =====================
# 4. TRANSFORMER MODEL
# =====================
def build_transformer(heads, dim, features):
    enc_in = layers.Input((INPUT_LEN, features))
    dec_in = layers.Input((HORIZON, features))

    attn = layers.MultiHeadAttention(heads, dim)(enc_in, enc_in)
    x = layers.LayerNormalization()(enc_in + attn)

    ff = layers.Dense(64, activation="relu")(x)
    enc_out = layers.LayerNormalization()(x + layers.Dense(features)(ff))

    self_attn = layers.MultiHeadAttention(heads, dim)(dec_in, dec_in)
    x = layers.LayerNormalization()(dec_in + self_attn)

    cross_attn = layers.MultiHeadAttention(heads, dim)(x, enc_out)
    x = layers.LayerNormalization()(x + cross_attn)

    ff = layers.Dense(64, activation="relu")(x)
    out = layers.Dense(features)(ff)

    model = models.Model([enc_in, dec_in], out)
    model.compile(optimizer="adam", loss="mse")
    return model

# =====================
# 5. HYPERPARAMETER SEARCH
# =====================
best_model, best_loss = None, np.inf
search_space = [(2, 16), (4, 32)]

for heads, dim in search_space:
    model = build_transformer(heads, dim, X_train.shape[2])
    history = model.fit(
        [X_train, decoder_train],
        y_train,
        epochs=10,
        batch_size=32,
        validation_split=0.1,
        verbose=0
    )
    val_loss = min(history.history["val_loss"])
    if val_loss < best_loss:
        best_loss = val_loss
        best_model = model

transformer = best_model

# =====================
# 6. LSTM BASELINE
# =====================
lstm = models.Sequential([
    layers.LSTM(64, input_shape=(INPUT_LEN, X.shape[2])),
    layers.Dense(HORIZON * X.shape[2]),
    layers.Reshape((HORIZON, X.shape[2]))
])

lstm.compile(optimizer="adam", loss="mse")
lstm.fit(X_train, y_train, epochs=20, batch_size=32, verbose=0)

# =====================
# 7. MULTIVARIATE VAR BASELINE
# =====================
train_var = data.iloc[:split + INPUT_LEN]
test_var  = data.iloc[split + INPUT_LEN:]

var_model = VAR(train_var)
var_fit = var_model.fit(maxlags=5)
var_preds = var_fit.forecast(train_var.values[-5:], steps=len(test_var))

# =====================
# 8. EVALUATION
# =====================
def evaluate(y_true, y_pred, h):
    yt = y_true[:, h-1]
    yp = y_pred[:, h-1]
    return (
        mean_absolute_error(yt, yp),
        np.sqrt(mean_squared_error(yt, yp))
    )

tf_preds   = transformer.predict([X_test, decoder_test], verbose=0)
lstm_preds = lstm.predict(X_test, verbose=0)

# =====================
# 9. STANDALONE SUMMARY BLOCK (DELIVERABLE 3)
# =====================
print("\n================ FINAL SUMMARY =================\n")

for name, preds in {"Transformer": tf_preds, "LSTM": lstm_preds}.items():
    mae, rmse = evaluate(y_test, preds, 10)
    print(f"{name} → Horizon 10 | MAE={mae:.4f}, RMSE={rmse:.4f}")

var_rmse = np.sqrt(mean_squared_error(test_var.values, var_preds))
print(f"VAR (Multivariate Classical Baseline) → RMSE={var_rmse:.4f}")

print("\n===============================================\n")
